extern "C" int format_timestamp_line(
  char* out, size_t out_size, const Timestamp* t,
  uint8_t PLACES, uint8_t WRAP_SECONDS, const char* ch_label);

void setup() {
  Serial.begin(115200);
  init_timestamp_state();
  // configure timers/ISRs...
}

void loop() {
  // Snapshot events atomically
  bool do0 = false, do1 = false;
  uint64_t pic0 = 0, tof0 = 0;
  uint64_t pic1 = 0, tof1 = 0;

  noInterrupts();
  if (event_ready_ch0) {
    event_ready_ch0 = false;
    pic0 = ch[0].picstop_latched;  // tick count at event
    tof0 = latest_tof_ps_ch0;      // absolute tof (ps) for this event
    do0 = true;
  }
  if (event_ready_ch1) {
    event_ready_ch1 = false;
    pic1 = ch[1].picstop_latched;
    tof1 = latest_tof_ps_ch1;
    do1 = true;
  }
  interrupts();

  if (do0) {
    update_timestamp_from_event(ch[0], pic0, tof0);
    char buf[64];
    int n = format_timestamp_line(buf, sizeof(buf), &ch[0].ts, 12, 2, "ch0");
    if (n > 0) Serial.write(buf, n);
  }

  if (do1) {
    update_timestamp_from_event(ch[1], pic1, tof1);
    char buf[64];
    int n = format_timestamp_line(buf, sizeof(buf), &ch[1].ts, 12, 2, "ch1");
    if (n > 0) Serial.write(buf, n);
  }
}
